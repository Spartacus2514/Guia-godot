Tu script de datos, completo

Así queda datos_granja.gd con todo el módulo. Compáralo con el tuyo. En el módulo 5 lo convertiremos en la economía real del juego.

extends Node

signal dinero_cambiado(nuevo_dinero: int)
signal energia_agotada

const PRECIOS = {"nabo": 20, "patata": 35, "fresa": 60}
const MAX_ENERGIA = 100

@export var nombre_granja: String = "Valle Verde"
@export var dinero: int = 500

var energia: int = MAX_ENERGIA
var inventario: Dictionary = {}
var cultivos_plantados: Array[String] = []


func _ready() -> void:
	dinero_cambiado.connect(_on_dinero_cambiado)
	energia_agotada.connect(_on_energia_agotada)
	print("Bienvenido a ", nombre_granja)

	comprar_semillas("nabo", 3)
	plantar("nabo")
	plantar("nabo")
	print("Inventario: ", inventario)
	print("Plantados: ", cultivos_plantados)


func puede_comprar(precio: int) -> bool:
	return dinero >= precio


func comprar_semillas(cultivo: String, cantidad: int) -> void:
	var total: int = PRECIOS[cultivo] * cantidad
	if not puede_comprar(total):
		print("No te llega el dinero")
		return
	dinero -= total
	dinero_cambiado.emit(dinero)
	agregar_objeto("semilla_" + cultivo, cantidad)


func agregar_objeto(nombre: String, cantidad: int) -> void:
	inventario[nombre] = inventario.get(nombre, 0) + cantidad


func quitar_objeto(nombre: String, cantidad: int) -> bool:
	if inventario.get(nombre, 0) < cantidad:
		return false
	inventario[nombre] -= cantidad
	if inventario[nombre] == 0:
		inventario.erase(nombre)
	return true


func plantar(cultivo: String) -> void:
	if not quitar_objeto("semilla_" + cultivo, 1):
		print("No tienes semillas de ", cultivo)
		return
	cultivos_plantados.append(cultivo)
	gastar_energia(5)


func gastar_energia(cantidad: int) -> void:
	energia -= cantidad
	if energia <= 0:
		energia = 0
		energia_agotada.emit()


func _on_dinero_cambiado(nuevo_dinero: int) -> void:
	print("Dinero: ", nuevo_dinero)


func _on_energia_agotada() -> void:
	print("¡Te desmayas!")
